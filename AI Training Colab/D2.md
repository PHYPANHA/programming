import os
import shutil
import zipfile
import glob

# ==============================================================
# STEP 1: Path Configuration
# ==============================================================
DATA_DIR = "/content/drive/MyDrive/rice_pest_project"
RAW_DIR = "/content/paddy_raw"  # Target local directory for clean extraction
MAIN_ZIP = os.path.join(DATA_DIR, "Paddy.zip")

# ==============================================================
# STEP 2: Directory Cleanup & Initialization
# ==============================================================
# Delete the existing directory if it exists to remove scrambled or duplicate folders
if os.path.exists(RAW_DIR):
    print(f"[INFO] Removing existing {RAW_DIR} directory for a fresh start...")
    shutil.rmtree(RAW_DIR)

# Create a brand new, empty target directory
os.makedirs(RAW_DIR, exist_ok=True)

# ==============================================================
# STEP 3: Extract Main ZIP File
# ==============================================================
if not os.path.exists(MAIN_ZIP):
    print(f"[ERROR] Paddy.zip not found at: {MAIN_ZIP}")
else:
    print("Step 1: Extracting the main Paddy.zip...")
    with zipfile.ZipFile(MAIN_ZIP, "r") as z:
        z.extractall(RAW_DIR)
    print(f" -> Success: Main ZIP extracted to {RAW_DIR}")

    # ==============================================================
    # STEP 4: Recursive Extraction for Nested ZIP Files
    # ==============================================================
    iteration = 1
    while True:
        # Search for any remaining .zip files inside RAW_DIR recursively
        current_zips = glob.glob(os.path.join(RAW_DIR, "**", "*.zip"), recursive=True)

        # Break the loop if no more nested ZIP files are found
        if not current_zips:
            print("\n[SUCCESS] All nested ZIP files have been fully and cleanly extracted!")
            break

        print(f"\n[Iteration {iteration}] Found {len(current_zips)} nested ZIP file(s) to extract.")

        for iz in current_zips:
            base_name = os.path.splitext(os.path.basename(iz))[0]

            # Apply naming convention: split by hyphen if present
            folder_name = base_name.split("-")[0] if "-" in base_name else base_name

            # Define extraction path directly relative to the current zip file location
            extract_to = os.path.join(os.path.dirname(iz), folder_name)
            os.makedirs(extract_to, exist_ok=True)

            try:
                with zipfile.ZipFile(iz, "r") as z:
                    z.extractall(extract_to)
                print(f"  Done: {os.path.basename(iz)} -> Extracted to folder '{folder_name}'")

                # Delete the inner zip file after successful extraction to prevent infinite loops
                os.remove(iz)
            except zipfile.BadZipFile:
                print(f"  [WARNING] Corrupted file detected and removed: {iz}")
                os.remove(iz)

        iteration += 1

def print_tree(path, prefix="", max_depth=3, current_depth=0):
    """Print directory tree with file counts."""
    if current_depth >= max_depth:
        return

    entries = sorted(os.listdir(path))
    dirs = [e for e in entries if os.path.isdir(os.path.join(path, e))]
    files = [e for e in entries if os.path.isfile(os.path.join(path, e))]

    if files:
        print(f"{prefix}[{len(files)} files]")

    for i, d in enumerate(dirs):
        connector = "+-- " if i == len(dirs) - 1 else "|-- "
        print(f"{prefix}{connector}{d}/")
        extension = "    " if i == len(dirs) - 1 else "|   "
        print_tree(os.path.join(path, d), prefix + extension,
                   max_depth, current_depth + 1)

for subdir in ["original_data", "augmented_data", "preprocessed_data"]:
    full_path = os.path.join(RAW_DIR, subdir)
    if os.path.exists(full_path):
        print("\n" + "= * 0")
        print(f"  {subdir}")
        print("= * 0")
        print_tree(full_path)
    else:
        print(f"  WARNING: {subdir} not found. Check extraction.")
