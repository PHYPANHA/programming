```py
# ==========================================
# STEP 2: Data Pre-processing & Merging — UNFIXED VERSION
# ==========================================
# 1. Strip hidden whitespace from all columns first
for df in [df_fao_raw, df_rain_raw, df_temp_raw, df_pest_raw]:
    df.columns = df.columns.str.strip()

# 2. Schema alignment: handle Kaggle's inconsistent temp.csv columns
if 'country' in df_temp_raw.columns:
    df_temp_raw.rename(columns={'country': 'Area'}, inplace=True)
if 'year' in df_temp_raw.columns:
    df_temp_raw.rename(columns={'year': 'Year'}, inplace=True)

# 3. Filter each source to Cambodia
df_yield = df_fao_raw[
    (df_fao_raw['Area'].str.lower() == 'cambodia') &
    (df_fao_raw['Item'].str.strip() == 'Rice') &
    (df_fao_raw['Element'].str.strip() == 'Yield')
][['Year', 'Value']].rename(columns={'Value': 'Yield_hg_ha'})

df_rain = df_rain_raw[df_rain_raw['Area'].str.lower() == 'cambodia'][['Year', 'average_rain_fall_mm_per_year']] \
    .rename(columns={'average_rain_fall_mm_per_year': 'Rainfall_mm'})

df_temp = df_temp_raw[df_temp_raw['Area'].str.lower() == 'cambodia'][['Year', 'avg_temp']] \
    .rename(columns={'avg_temp': 'Avg_Temp'})

df_pest = df_pest_raw[df_pest_raw['Area'].str.lower() == 'cambodia'][['Year', 'Value']] \
    .rename(columns={'Value': 'Pesticides_tonnes'})

# 4. Merge all sources on Year
yield_df = df_yield.merge(df_rain, on='Year', how='left') \
                   .merge(df_temp, on='Year', how='left') \
                   .merge(df_pest, on='Year', how='left')

# Fill missing values with 0 — a quick "fix" (what could go wrong?)
yield_df['Avg_Temp'] = yield_df['Avg_Temp'].fillna(0.0)
yield_df['Pesticides_tonnes'] = yield_df['Pesticides_tonnes'].fillna(0.0)

yield_df_final = yield_df.dropna()

print(f"Merged rows: {len(yield_df_final)}")
if len(yield_df_final) > 0:
    print(f"Year range: {yield_df_final['Year'].min()} - {yield_df_final['Year'].max()}")
```

```py
# ==========================================
# STEP 4: Model Training, Evaluation, and Visual Output
# ==========================================
models = {
    "Linear Regression": LinearRegression(),
    "Random Forest": RandomForestRegressor(n_estimators=30, max_depth=3, random_state=42),
    "XGBoost": XGBRegressor(n_estimators=30, max_depth=2, learning_rate=0.1, random_state=42)
}

results = []
best_r2 = -np.inf
best_model_name = ""
best_model_obj = None

print("\n--- Model Training & Evaluation ---")
for name, model in models.items():
    model.fit(X_train, y_train)
    predictions = model.predict(X_test)
    r2 = r2_score(y_test, predictions)
    rmse = np.sqrt(mean_squared_error(y_test, predictions))
    results.append({'Model': name, 'Test R2': round(r2, 4), 'Test RMSE': round(rmse, 2)})
    print(f"  {name}: Test R² = {r2:.4f}")
    if r2 > best_r2:
        best_r2 = r2
        best_model_name = name
        best_model_obj = model

performance_df = pd.DataFrame(results).sort_values(by='Test R2', ascending=False)
print("\n--- Performance Comparison ---")
display(performance_df)
print(f"\nBest Model Selected: {best_model_name} (Test R² = {best_r2:.4f})")
```
